    <script>
        let tokenClient;
        let accessToken = null;

        // 1. 初始化 Google 客戶端
        function initClient() {
            tokenClient = google.accounts.oauth2.initTokenClient({
                client_id: '1018971008023-u5tooicug7ij968fiibn7fk7tf0ubc58.apps.googleusercontent.com', // <-- 填入你的 Client ID
                scope: 'https://www.googleapis.com/auth/drive.appdata', // 我們剛設定的權限
                callback: (tokenResponse) => {
                    if (tokenResponse && tokenResponse.access_token) {
                        accessToken = tokenResponse.access_token;
                        document.getElementById('status').innerHTML = 
                            '<b>登入成功！</b><br>你的 Access Token 是：<br>' + accessToken;
                        console.log("Token:", accessToken);
                        // 這裡拿到 Token 後，就可以開始呼叫 Google Drive API 讀寫資料了
                    }
                },
            });
        }

        // 2. 點擊按鈕觸發登入
        function handleAuthClick() {
            // 請求 Token
            tokenClient.requestAccessToken();
